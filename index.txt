function initNet(room) {
  roomRef = db.ref('tag/' + mode + '/' + $('map').value + '/' + room); itRef = roomRef.child('it');
  const my = roomRef.child('players/' + me);
  my.onDisconnect().remove();

  roomRef.child('players').on('value', s => {
    players = s.val() || {};
    if (mode == 'tag' && (!itId || !players[itId])) claimIt();
  });
  itRef.on('value', s => {
    const v = s.val();
    if (v) { if (v.id === me && itId !== me && navigator.vibrate) navigator.vibrate(300); itId = v.id; tagT = v.t; }
    else itId = null;
  });
  roomRef.child('seed').on('value', sn => {
    const v = sn.val(); if (v == null) { sn.ref.transaction(c => c == null ? 0 : undefined); return; }
    if (v !== seed0) { seed0 = v; if (mode == 'killer') startRound(); }
  });
  roomRef.child('hits/' + me).on('child_added', sn => {
    const v = sn.val(); sn.ref.remove();
    if (v.kill) creditMe(); else hurt(v.d, v.f);
  });
  roomRef.child('fx').on('child_added', sn => {
    const v = sn.val(); if (v.f === me) return;
    addFx(v.k, new THREE.Vector3(...v.a), v.b ? new THREE.Vector3(...v.b) : null, v.r, true);
  });
  setInterval(() => { // отправка позиции 10 раз/с
    roomRef.child('players/' + me).update({x:+pos.x.toFixed(2), z:+pos.z.toFixed(2), r:+yawOf().toFixed(2), hp:Math.round(hp), k:kills, dd:dead ? 1 : 0, ro:myRole});
  }, 100);
}
function claimIt() {
  itRef.transaction(cur => (!cur || !players[cur.id]) ? {id: me, t: Date.now()} : undefined);
}
function yawOf() {

}

  hp -= d; $('flash').style.opacity = .5; setTimeout(() => $('flash').style.opacity = 0, 120);
  if (navigator.vibrate) navigator.vibrate(80);
  if (hp <= 0) {
    hp = 0; dead = Date.now();
    if (String(from).startsWith('bot')) players[from].k = (players[from].k||0) + 1;
    else if (from && !offline) roomRef.child('hits/' + from).push({kill:1});
  }
}
function fire() {
  const T = TOOLS[tool], t0 = performance.now()/1000;
  if (!T || dead || winner || Date.now() < hold || t0 - lastFire < T.cd) return; lastFire = t0;
  const o = new THREE.Vector3(pos.x, 1.5, pos.z), dir = new THREE.Vector3(0,0,-1).applyQuaternion(headQ), fl = Math.hypot(dir.x, dir.z) || 1;
  if (tool == 'gun' || tool == 'revolver') {
    let best = null, bt = 40;
    for (const id in players) {
      const q = players[id]; if (id == me || q.dd || q.x == null) continue;
      const c = new THREE.Vector3(q.x, 1.1, q.z).sub(o), t = c.dot(dir);
      if (t > 0 && t < bt && c.addScaledVector(dir, -t).length() < .6) { bt = t; best = id; }
    }
    obstacles.forEach(ob => { const cx = ob.x-o.x, cz = ob.z-o.z, t = (cx*dir.x + cz*dir.z)/(fl*fl);
      if (t > 0 && t < bt && Math.hypot(cx - dir.x*t, cz - dir.z*t) < ob.r) { best = null; bt = t; } });
    addFx('line', o.clone().add(new THREE.Vector3(0,-.2,0)), o.clone().addScaledVector(dir, bt), 0);
    if (best) hit(best, T.d);
  } else if (tool == 'bazooka') {
    const m = new THREE.Mesh(new THREE.SphereGeometry(.12, 8, 8), new THREE.MeshBasicMaterial({color:0xff5522})); scene.add(m);
    projs.push({p: o.clone().addScaledVector(dir, .8), v: dir.clone().multiplyScalar(18), m});
  } else {
    for (const id in players) {
      const q = players[id]; if (id == me || q.dd || q.x == null) continue;
      const dx = q.x - pos.x, dz = q.z - pos.z, dd = Math.hypot(dx, dz) || 1;
      if (dd < T.r && (dx*dir.x + dz*dir.z)/(dd*fl) > .5) { hit(id, T.d); addFx('boom', new THREE.Vector3(q.x, 1.2, q.z), null, .6); }
    }
  }
}
function explode(p) {
  addFx('boom', p, null, 3);
  for (const id in players) {
    const q = players[id]; if (id == me || q.dd || q.x == null) continue;
    const d = Math.hypot(p.x - q.x, p.z - q.z); if (d < 3.2) hit(id, TOOLS.bazooka.d * (1 - d/3.5));
  }
}
function gameTick(dt) {
  const now = Date.now();
  if (selfAv) { selfAv.position.set(pos.x, 0, pos.z); const fw = new THREE.Vector3(0,0,-1).applyQuaternion(headQ); selfAv.rotation.y = Math.atan2(-fw.x, -fw.z); }
  if (mode == 'duel' || mode == 'killer') {
    const cnt = Object.keys(players).length;
    if (cnt != lastCnt) { lastCnt = cnt; if (mode == 'duel') duelReset(); else startRound(); }
    if (kills != lastKills) { lastKills = kills; if (mode == 'duel') resetAt = now + 3000; }
    if (resetAt && now > resetAt && !winner) { resetAt = 0; duelReset(); }
    if (mode == 'killer') roundTick(now);
  }
  if (gest == 'peace' && now > dashCd && !dead) { dashT = now + 800; dashCd = now + 5000; }
  fxs.forEach((f, i) => { const a = (performance.now() - f.t)/1000, life = f.k == 'line' ? .2 : .5;
    if (f.k != 'line') { f.m.scale.setScalar(1 + a*2); f.m.material.opacity = Math.max(0, 1 - a/life); }
    if (a > life) { scene.remove(f.m); fxs.splice(i, 1); } });
  if (mode == 'tag') return;
  const f = new THREE.Vector3(0,0,-1).applyQuaternion(headQ); f.y = 0; f.normalize();
  const n = slots.length;
  slots.forEach((sl, i) => { const o = (i - (n-1)/2) * .2; sl.position.set(pos.x + f.x*.3 - f.z*o, 1.1, pos.z + f.z*.3 + f.x*o); sl.rotation.y += dt; });
  head.updateMatrixWorld(true);
  if (handGroup.visible) { const w = handDots[9].getWorldPosition(new THREE.Vector3()); slots.forEach(sl => { if (w.distanceTo(sl.position) < .17) equip(sl.userData.t); }); }
  if (toolMesh) { toolMesh.visible = handGroup.visible; toolMesh.position.copy(handDots[9].position); }
  if (gest == 'point') fire();
  if (gest == 'rock' && now > swCd && slots.length) { swCd = now + 700; const ts = [...MODES[mode][1], null]; equip(ts[(ts.indexOf(tool) + 1) % ts.length]); }
  for (let i = projs.length - 1; i >= 0; i--) {
    const q = projs[i]; q.p.addScaledVector(q.v, dt); q.m.position.copy(q.p);
    const bang = q.p.y < .1 || Math.abs(q.p.x) > ARENA || Math.abs(q.p.z) > ARENA || obstacles.some(o => Math.hypot(q.p.x-o.x, q.p.z-o.z) < o.r)
      || Object.keys(players).some(id => id != me && !players[id].dd && players[id].x != null && Math.hypot(q.p.x-players[id].x, q.p.z-players[id].z) < .7);
    if (bang) { explode(q.p); scene.remove(q.m); projs.splice(i, 1); }
  }
  if (dead && now - dead > 3000 && !winner && mode != 'killer') { if (mode == 'duel') { myDeaths++; duelReset(); } else { dead = 0; hp = 100; pos.set((Math.random()-.5)*ARENA*1.4, 0, (Math.random()-.5)*ARENA*1.4); } }
  if (offline) players[me].k = kills;
  if (!winner) { for (const id in players) if ((id == me ? kills : players[id].k||0) >= KILLS) { winner = players[id].n; winT = now; } }
  else if (now - winT > 6000) {
    winner = null; kills = 0; myDeaths = 0; hp = 100; dead = 0;
    if (offline) BOTS.forEach(id => Object.assign(players[id], {k:0, hp:100, dd:0}));
  }
}
function botFight(id, dt) {
  const b = players[id], now = Date.now(), lvl = +$('diff').value, role = b.ro;
  if (b.dd) { if (mode != 'killer' && mode != 'duel' && now - b.dd > 3000) { b.dd = 0; b.hp = 100; b.x = pos.x > 0 ? -ARENA*.7 : ARENA*.7; b.z = pos.z > 0 ? -ARENA*.7 : ARENA*.7; } return; }
  if (winner || roundEnd || (mode != 'killer' && dead) || now < hold + [900, 550, 300][lvl]) return;
  let tg = null, td = 1e9;
  for (const k in players) {
    if (k == id || isDead(k)) continue;
    const q = players[k], qx = k == me ? pos.x : q.x, qz = k == me ? pos.z : q.z; if (qx == null) continue;
    if (mode == 'killer' && role != 'killer' && roleOf(k) != 'killer') continue; // шериф и выжившие следят за убийцей
    const dd = Math.hypot(qx - b.x, qz - b.z); if (dd < td) { td = dd; tg = {k, x: qx, z: qz}; }
  }
  if (!tg) return;
  const T = TOOLS[b.t], dx = tg.x - b.x, dz = tg.z - b.z, d = Math.hypot(dx, dz) || 1;
  const surv = mode == 'killer' && role == 'survivor', pref = T.r ? 1.2 : b.t == 'gun' ? 7 : b.t == 'revolver' ? 8 : 10;
  let sv = surv ? (d < 12 ? -1 : 0) : d > pref ? 1 : d < pref - 1.5 ? -1 : 0;
  if (mode == 'duel') sv = 0;
  const st = mode == 'duel' ? 0 : Math.sin(now/900 + (id == 'bot2' ? 2 : 0)) * .6;
  let vx = dx/d*sv - dz/d*st, vz = dz/d*sv + dx/d*st;
  const l = Math.hypot(vx, vz) || 1; vx /= l; vz /= l;
  b.x += vx*botSpeed*.8*dt; b.z += vz*botSpeed*.8*dt;
  b.x = Math.max(-ARENA+.6, Math.min(ARENA-.6, b.x)); b.z = Math.max(-ARENA+.6, Math.min(ARENA-.6, b.z));
  obstacles.forEach(o => { const ox = b.x-o.x, oz = b.z-o.z, od = Math.hypot(ox, oz), min = o.r + .4;
    if (od < min && od > 0) { b.x = o.x + ox/od*min; b.z = o.z + oz/od*min; } });
  b.r = Math.atan2(-dx, -dz);
  if (!surv && now > b.next && d < (T.r || 20)) {
    b.next = now + T.cd*1000*[2.2, 1.5, 1.1][lvl] + 300;
    const from = new THREE.Vector3(b.x, 1.4, b.z), to = new THREE.Vector3(tg.x, 1.2, tg.z);
    if (b.t == 'gun' || b.t == 'revolver') addFx('line', from, to, 0);
    if (b.t == 'bazooka') addFx('boom', to, null, 3);
    if (Math.random() < [.3, .5, .75][lvl]) {
      const dm = (mode == 'killer' || mode == 'duel') ? 100 : T.d * [.6, .8, 1][lvl];
      if (tg.k == me) hurt(dm, id);
      else { const q = players[tg.k]; q.hp -= dm; if (q.hp <= 0) { q.dd = now; b.k = (b.k||0) + 1; } }
    }
  }
}

function startRound() {
  const ids = offline ? [me, ...BOTS] : Object.keys(players).sort();
  const P = [['killer','sheriff','survivor'],['sheriff','survivor','killer'],['survivor','killer','sheriff'],['killer','survivor','sheriff'],['sheriff','killer','survivor'],['survivor','sheriff','killer']];
  const sd = offline ? Math.random()*6|0 : seed0;
  const rl = ids.length < 3 ? (sd % 2 ? ['killer','sheriff'] : ['sheriff','killer']) : P[sd % 6];
  ids.forEach((id, i) => { const r = rl[i] || 'survivor';
    if (id == me) myRole = r;
    else if (offline) Object.assign(players[id], {ro: r, hp: 100, dd: 0, t: r == 'killer' ? 'knife' : r == 'sheriff' ? 'revolver' : 'fist',
      x: (Math.random()-.5)*ARENA*1.4, z: (Math.random()-.5)*ARENA*1.4});
  });
  hp = 100; dead = 0; roundEnd = null; roundStart = Date.now();
  pos.set((Math.random()-.5)*ARENA*1.4, 0, (Math.random()-.5)*ARENA*1.4);
  equip(myRole == 'killer' ? 'knife' : myRole == 'sheriff' ? 'revolver' : null);

function roundTick(now) {
  if (!myRole || now - roundStart < 3000 || (!offline && Object.keys(players).length < 2)) return;
  if (!roundEnd) {
    const al = Object.keys(players).filter(id => !isDead(id));
    const ka = al.some(id => roleOf(id) == 'killer'), oa = al.some(id => roleOf(id) != 'killer');
    const text = !ka ? '🏆 Убийцу поймали! Победили выжившие' : !oa ? '🔪 Убийца победил!' : now - roundStart > 120000 ? '⏰ Время вышло — победили выжившие' : '';
    if (text) roundEnd = {text, t: now, s: seed0};
  } else if (now - roundEnd.t > 5000) {
    if (offline) startRound();
    else { roundEnd.t = now; if (Object.keys(players).sort()[0] == me) roomRef.child('seed').transaction(c => c === roundEnd.s ? c + 1 : undefined); }
  }
}
function modeHud() {

  if (mode == 'killer') return roundEnd ? roundEnd.text + '\n' : '🎭 ' + (ROLEN[myRole] || '…') + ' · ⏱ ' + Math.max(0, 120 - ((now - roundStart)/1000|0)) + 'с\n';
  return '';
}

function updateBot(dt) {

  const b = players.bot, now = Date.now();

    if (d < 1.2) { setIt(me); if (navigator.vibrate) navigator.vibrate(300); return; }
  }
  let vx = chase ? dx/d : -dx/d, vz = chase ? dz/d : -dz/d;
  if (!chase) { // убегая, держится подальше от стен и тянется к центру
    const k = Math.max(Math.abs(b.x), Math.abs(b.z)) / ARENA;
    vx += -b.x/ARENA*k*k*3; vz += -b.z/ARENA*k*k*3;
    const t = now/700; vx += Math.sin(t)*.3; vz += Math.cos(t)*.3;
  }
  obstacles.forEach(o => { // огибает препятствия
    const ox = b.x-o.x, oz = b.z-o.z, od = Math.hypot(ox, oz) || 1;

}

function initHands(video) {
  const hands = new Hands({locateFile: f => 'https://cdn.jsdelivr.net/npm/@mediapipe/hands@0.4.1646424915/' + f});
  hands.setOptions({maxNumHands:1, modelComplexity:0, minDetectionConfidence:.6, minTrackingConfidence:.5});
  hands.onResults(r => {
    const lm = r.multiHandLandmarks && r.multiHandLandmarks[0];
    if (!lm) { handGroup.visible = false; moving = false; gest = 'none'; return; }
    handGroup.visible = true;
    const W = .84, H = .63, D = .6;
    lm.forEach((p, i) => handDots[i].position.set((p.x-.5)*W, (.5-p.y)*H, -D + p.z*-.5));
    const d = (a, b) => Math.hypot(lm[a].x-lm[b].x, lm[a].y-lm[b].y);
    const f = [[8,6],[12,10],[16,14],[20,18]].map(([t, p]) => d(t, 0) > d(p, 0)*1.1);
    const up = f.filter(Boolean).length, th = d(4,17) > d(3,17)*1.15;
    gest = up >= 3 ? 'open' : (f[0] && !f[1] && !f[2] && !f[3]) ? 'point' : (f[0] && f[1] && !f[2] && !f[3]) ? 'peace' : (f[0] && !f[1] && !f[2] && f[3]) ? 'rock' : up == 0 ? (th ? 'thumb' : 'fist') : 'none';
    moving = gest == 'open';
  });
  (async function pump() {
    if (video.readyState >= 2) { try { await hands.send({image: video}); } catch(e){} }
    requestAnimationFrame(pump);
  })();
}
